agent slime {
    i. pos(1,2)
    i. dir(right)
    i. state(idle)

    see(player) ->
        n. state(chase)

    state(chase) ->
        n. move_toward(player)

    low_health ->
        n. state(flee)
}
agent guard {
    i. pos(5,8)
    i. state(patrol)

    see(enemy) ->
        n. state(alert)

    state(alert) ->
        e. attack(enemy)

    state(patrol) u. see(enemy)

    a. health(0) -> state(dead)
}


rel see(enemy)
next alert=1
